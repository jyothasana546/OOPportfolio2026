package Week8;

public abstract class StaffMember {
	  protected String name;
	    protected String address;
	    protected String phone;

	    public StaffMember(String name, String address, String phone) {
	        this.name = name;
	        this.address = address;
	        this.phone = phone;
	    }

	    public abstract double pay();

	    // NEW: Vacation method
	    public int vacationDays() {
	        return 10; // default
	    }

	    public String toString() {
	        return name + "\n" + address + "\n" + phone;
	    }
}

package Week8;

public class Hourly  extends Employee {
    private int hoursWorked;
    private double rate;

    public Hourly(String name, String address, String phone, double rate) {
        super(name, address, phone, 0);
        this.rate = rate;
    }

    public void addHours(int hours) {
        hoursWorked += hours;
    }

    public double pay() {
        double payment = hoursWorked * rate;
        hoursWorked = 0;
        return payment;
    }

    public int vacationDays() {
        return 10;
    }
}

Child Classes -  Password,  Secret
package Week8;

public class Password extends Encryptable {
    private String password;

    public Password(String password) {
        this.password = password;
    }

    public void encrypt() {
        password = new StringBuilder(password).reverse().toString();
    }

    public String decrypt() {
        return new StringBuilder(password).reverse().toString();
    }

    public String toString() {
        return password;
    } 

}

package Week8;

public class Secret   extends Encryptable {
    private String message;

    public Secret(String msg) {
        message = msg;
    }

    public void encrypt() {
        message = message.toUpperCase(); // simple Caesar-like
    }

    public String decrypt() {
        return message.toLowerCase();
    }

    public String toString() {
        return message;
    }

}
