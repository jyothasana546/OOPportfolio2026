package Week9;
import java.util.Scanner;
public class Histogram {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);

        int[] frequency = new int[10];

        System.out.println("Enter numbers between 1 and 100 (enter 0 to stop):");

        while (true) {
            int num = sc.nextInt();

            
            if (num == 0) {
                break;
            }

            
            if (num >= 1 && num <= 100) {
                int index = (num - 1) / 10;
                frequency[index]++;
            } else {
                System.out.println("Invalid input! Enter between 1–100.");
            }
        }

        
        System.out.println("\nHistogram:\n");

        for (int i = 0; i < 10; i++) {
            int lower = i * 10 + 1;
            int upper = (i + 1) * 10;

            System.out.printf("%2d - %3d | ", lower, upper);

            for (int j = 0; j < frequency[i]; j++) {
                System.out.print("*");
            }

            System.out.println();
        }

        sc.close();
    }
}

package Week9;

public class Bank {
	private Account[] accounts;
    private int count;

    public Bank() {
        accounts = new Account[30]; // max 30 customers
        count = 0;
    }

    public void createAccount(int accNo, String name, double balance) {
        if (count < 30) {
            accounts[count] = new Account(accNo, name, balance);
            count++;
            System.out.println("Account created successfully.");
        } else {
            System.out.println("Bank is full! Cannot add more accounts.");
        }
    }

    private Account findAccount(int accNo) {
        for (int i = 0; i < count; i++) {
            if (accounts[i].getAccNumber() == accNo) {
                return accounts[i];
            }
        }
        return null;
    }

    public void deposit(int accNo, double amount) {
        Account acc = findAccount(accNo);
        if (acc != null) {
            acc.deposit(amount);
        } else {
            System.out.println("Account not found!");
        }
    }

    public void withdraw(int accNo, double amount) {
        Account acc = findAccount(accNo);
        if (acc != null) {
            acc.withdraw(amount);
        } else {
            System.out.println("Account not found!");
        }
    }

    public void addInterestToAll() {
        for (int i = 0; i < count; i++) {
            accounts[i].addInterest();
        }
        System.out.println("3% interest added to all accounts.");
    }

    public void displayAll() {
        for (int i = 0; i < count; i++) {
            accounts[i].display();
        }
    }


}

        Person[] list = new Person[25];

        while (count < 25) {
            String line = sc.nextLine();

            if (line.equalsIgnoreCase("stop")) {
                break;
            }

            // Split by TAB
            String[] parts = line.split("\t");

            if (parts.length == 3) {
                list[count] = new Person(parts[0], parts[1], parts[2]);
                count++;
            } else {
                System.out.println("Invalid input! Please use TAB between values.");
            }
        }

        // Display output
        System.out.println("\nStored Data:\n");
        System.out.printf("%-15s %-15s %-10s\n", "First Name", "Last Name", "Postal Code");
        System.out.println("---------------------------------------------");

        for (int i = 0; i < count; i++) {
            list[i].display();
        }

        sc.close();
    }

package Week9;

public class Question {
	private String questionText;
    private String answer;

    public Question(String questionText, String answer) {
        this.questionText = questionText;
        this.answer = answer;
    }

    public String getQuestion() {
        return questionText;
    }

    public boolean checkAnswer(String userAnswer) {
        return answer.equalsIgnoreCase(userAnswer);
    }

}

package Week9;
import java.util.Scanner;

public class Quiz {
	private Question[] questions;
    private int count;

    public Quiz() {
        questions = new Question[25];
        count = 0;
    }

    // Add question
    public void add(Question q) {
        if (count < 25) {
            questions[count] = q;
            count++;
        } else {
            System.out.println("Quiz is full!");
        }
    }

    // Conduct quiz
    public void giveQuiz() {
        Scanner sc = new Scanner(System.in);
        int score = 0;

        for (int i = 0; i < count; i++) {
            System.out.println("Q" + (i + 1) + ": " + questions[i].getQuestion());
            System.out.print("Your answer: ");
            String userAnswer = sc.nextLine();

            if (questions[i].checkAnswer(userAnswer)) {
                System.out.println("Correct!\n");
                score++;
            } else {
                System.out.println("Wrong!\n");
            }
        }

        // Final result
        System.out.println("Quiz Finished!");
        System.out.println("Your Score: " + score + "/" + count);
    }
}
